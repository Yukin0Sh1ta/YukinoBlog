var fs = require("fs");
var http = require("http");
const { json } = require("stream/consumers");

var server = http.createServer((req, res) => {
  res.writeHead(200, {
    "Content-type": "application/json",
  });
  var obj = {
    name: "张三",
    age: 20,
  };
  res.end(JSON.stringify(obj));
});

server.listen(3000, "127.0.0.1");
console.log("success");
