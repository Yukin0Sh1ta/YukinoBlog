var fs = require("fs");
//同步
let read = fs.readFileSync("./test.txt", "utf8");
console.log("读取成功");
fs.writeFileSync("write.txt", read);
console.log("写入成功");
console.log("hello1");
console.log("hello2");
//读取成功 写入成功 hello1 hello2

//异步
let read1 = fs.readFile("./test.txt", "utf8", (error, data) => {
  console.log("读取成功");
  fs.writeFile("write.txt", data, (error, data) => {
    console.log("写入成功");
  });
  console.log("hello1");
});
console.log("hello2");
//hello2 读取成功 hello1 写入成功
